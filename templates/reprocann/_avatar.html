{# Macro para renderizar un avatar consistente. Uso: {{ avatar(user, size=44, link=true, classes='post-avatar') }} - user: objeto User (debe tener .username y .profile) - size: tamaño en px (default 44) - link: si true, envuelve en un al perfil público - classes: clases CSS extra (post-avatar, comment-avatar, suggest-avatar, etc.) #} {% macro avatar(user, size=44, link=true, classes='', extra_style='') -%} {%- set p = user.profile if (user and user.profile) else none -%} {%- set img = p.avatar_path if (p and p.avatar_path) else none -%} {%- set initial = (user.username[0]|upper) if user else '?' -%} {%- if link and user -%} {%- if not img -%}{{ initial }}{%- endif -%} {%- else -%} {%- if not img -%}{{ initial }}{%- endif -%} {%- endif -%} {%- endmacro %}
